name: security-pipeline-demo

# Demonstration pipeline. Runs a REAL SAST scan (Bandit) on the intentionally-vulnerable
# sample_app/, plus an SSH compliance test against a throwaway container, then builds a
# risk register and uploads the evidence. Sample code only — not a real product.
#
# To activate: move this file to .github/workflows/security-pipeline.yml
# (requires the pushing token to have the "Workflows" permission).

on:
  push:
    branches: [ main ]
  workflow_dispatch:

permissions:
  contents: read

jobs:
  pipeline:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Install scanners + deps
        run: pip install openpyxl bandit

      - name: SAST scan (Bandit) on the sample application
        run: |
          python -m bandit -r sample_app -f json -o bandit.json -q || true
          python pipeline/normalize_bandit.py bandit.json > pipeline/scan_findings.json
          echo "Findings:"; python -c "import json;print(len(json.load(open('pipeline/scan_findings.json'))))"

      - name: SSH compliance test (ephemeral key + throwaway container)
        run: |
          chmod +x scripts/ssh_compliance_test.sh
          scripts/ssh_compliance_test.sh "$GITHUB_WORKSPACE"

      - name: Pipeline — scan results -> ledger -> register -> verify
        run: |
          python pipeline/run_pipeline.py \
            --findings pipeline/scan_findings.json \
            --host-results "$GITHUB_WORKSPACE/host_results.json" \
            --out-dir "$GITHUB_WORKSPACE"

      - name: Upload evidence (ledger + register + raw scan)
        uses: actions/upload-artifact@v4
        with:
          name: pipeline-evidence
          path: |
            ledger.jsonl
            register.xlsx
            bandit.json
            host_results.json
